function submitTicket(e) { e.preventDefault(); const cat = document.getElementById('selectedCategory').value; const ign = document.getElementById('ign').value; const baleId = document.getElementById('baleId').value; const message = document.getElementById('ticketMessage').value; const code = "#NXS-" + Math.floor(1000 + Math.random() * 9000); document.getElementById('generatedCode').innerText = code; let targetUrl = "http://ble.ir/Modak"; let department = "منیجری و ادمین‌ها"; let catPersian = "پیشنهاد / ایده"; if (cat === 'test') { targetUrl = "http://ble.ir/amirldytuiop"; department = "تستر رسمی نکسوس (@amirldytuiop)"; catPersian = "تست ورودی کلن"; } else if (cat === 'staff' || cat === 'complaint') { targetUrl = "http://ble.ir/Im_himora"; department = "لیدر ارشد (@Im_himora)"; catPersian = cat === 'staff' ? "درخواست عضویت در تیم استف" : "شکایت و انتقاد"; } else if (cat === 'report') { targetUrl = "http://ble.ir/Modak"; department = "کادر نظارت و منیجر (@Modak)"; catPersian = "گزارش تخلف بازیکن"; } // تولید متن شکیل تیکت برای ارسال در بله const ticketText = `❖ تیکت رسمی نکسوس ❖\n` + `شناسه رهگیری: ${code}\n` + `موضوع: ${catPersian}\n` + `نام درون بازی (IGN): ${ign}\n` + `آیدی بله: ${baleId}\n` + `متن پیام: ${message}`; // کپی خودکار در کلیپ‌بورد کاربر navigator.clipboard.writeText(ticketText).then(() => { alert("✅ مشخصات تیکت در کلیپ‌بورد کپی شد! در صفحه بله فقط Paste (چسباندن) کنید و بفرستید."); }); document.getElementById('ticketSummary').innerHTML = `
دپارتمان: ${catPersian}
نام کاربری: ${ign}
آیدی ثبت‌کننده: ${baleId}
کد رهگیری: ${code}
`; document.getElementById('targetNotice').innerText = `تیکت شما به ${department} متصل شد. روی دکمه زیر کلیک کنید و پیام کپی‌شده را ارسال کنید.`; document.getElementById('baleLink').href = targetUrl; document.getElementById('ticket-box').style.display = 'block'; document.getElementById('submitBtn').style.display = 'none'; document.getElementById('ticket-box').scrollIntoView({ behavior: 'smooth' }); }
